/** * 노즈가드 임상연구 — 접수 API (Google Apps Script 웹 앱) * * 역할 * - POST : index.html 이 보내는 참여 신청·문의(JSON)를 구글 시트에 저장 * - GET : admin.html 이 관리자 키와 함께 요청하면 저장된 내역을 JSON 으로 반환 * * 설치·반영 * 1) 응답 저장용 구글 시트에서 [확장 프로그램 > Apps Script] 로 연 프로젝트에 이 파일 내용을 붙여넣는다. * (시트와 별개로 만든 프로젝트라면 아래 SPREADSHEET_ID 에 시트 주소의 ID 를 넣는다.) * 2) [프로젝트 설정(톱니) > 스크립트 속성 > 스크립트 속성 추가] * 속성: ADMIN_KEY 값: 16자 이상의 임의 문자열 (관리자 페이지 비밀번호 역할, 코드에 적지 말 것) * 3) [배포 > 배포 관리 > 연필(수정) > 버전: 새 버전 > 배포] * 다음 사용자로 실행: 나 / 액세스 권한이 있는 사용자: 모든 사용자 * 기존 배포를 '수정'하면 웹 앱 URL 이 그대로 유지된다. ('새 배포'를 만들면 URL 이 바뀐다.) * * 접수를 마감하려면 ACCEPTING 을 false 로 바꾸고 새 버전으로 배포한다. */ const ACCEPTING = true; const SPREADSHEET_ID = ''; // 시트에 연결된 프로젝트면 비워 둔다. const SHEET_NAME = '신청내역'; const HEADERS = ['접수일시', '성명', '소속 기관', '사업자등록번호', '연락처', '성별', '개인정보 동의', '상태']; const GENDERS = ['남성', '여성', '응답하지 않음']; const INQUIRY_SHEET_NAME = '문의내역'; const INQUIRY_HEADERS = ['접수일시', '성명', '연락처/이메일', '문의 내용', '처리상태']; // 새 신청·문의가 들어오면 알림 메일을 받을 주소. 비워 두면 보내지 않는다. (메일에는 개인정보를 넣지 않는다.) const NOTIFY_EMAIL = ''; /* ───────────── 웹 앱 진입점 ───────────── */ // GET ?action=list&key=관리자키 → 관리자 조회 function doGet(e) { const prm = (e && e.parameter) || {}; if (prm.action === 'list') return json_(adminList_(prm.key)); return json_({ ok: true, service: 'noseguard-intake' }); // 상태 확인용. 데이터는 반환하지 않는다. } // POST (본문: JSON 문자열) { type: 'application' | 'inquiry', ... } function doPost(e) { let p; try { p = JSON.parse(e.postData.contents); } catch (err) { return json_(fail_('요청 형식이 올바르지 않습니다.')); } if (!p || typeof p !== 'object') return json_(fail_('요청 형식이 올바르지 않습니다.')); if (p.type === 'application') return json_(submitApplication(p)); if (p.type === 'inquiry') return json_(submitInquiry(p)); return json_(fail_('요청 유형이 올바르지 않습니다.')); } /* ───────────── 접수 ───────────── */ function submitApplication(p) { if (!ACCEPTING) return fail_('현재 참여 신청 접수가 종료되었습니다.'); p = p || {}; if (p.website) return { ok: true }; // 봇이 채우는 숨김 입력: 성공처럼 응답하고 저장하지 않는다. const name = clean_(p.name, 50); const institution = clean_(p.institution, 100); const bizno = clean_(p.bizno, 12); const phone = clean_(p.phone, 20); const gender = clean_(p.gender, 20); if (!name || !institution) return fail_('성명과 소속 기관을 입력해 주세요.'); if (!/^\d{3}-\d{2}-\d{5}$/.test(bizno)) return fail_('사업자등록번호는 123-45-67890 형식으로 입력해 주세요.'); if (!/^01[016789]-?\d{3,4}-?\d{4}$/.test(phone)) return fail_('연락처는 010-1234-5678 형식으로 입력해 주세요.'); if (GENDERS.indexOf(gender) < 0) return fail_('성별을 선택해 주세요.'); if (p.consent !== true) return fail_('개인정보 수집·이용에 동의해야 신청할 수 있습니다.'); const lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (e) { return fail_('접속자가 많아 처리하지 못했습니다. 잠시 후 다시 시도해 주세요.'); } try { appendRow_(SHEET_NAME, HEADERS, [name, institution, bizno, phone, gender, '동의', '접수']); } catch (err) { console.error(err); return fail_('저장 중 오류가 발생했습니다. 잠시 후 다시 시도해 주세요.'); } finally { lock.releaseLock(); } notify_('새 참여 신청이 접수되었습니다.'); return { ok: true }; } function submitInquiry(p) { if (!ACCEPTING) return fail_('현재 문의 접수가 종료되었습니다.'); p = p || {}; if (p.website) return { ok: true }; const name = clean_(p.name, 50); const contact = clean_(p.contact, 100); const body = String(p.body == null ? '' : p.body) .replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f]/g, ' ').trim().slice(0, 1000); if (!name) return fail_('성명을 입력해 주세요.'); const isPhone = /^01[016789]-?\d{3,4}-?\d{4}$/.test(contact); const isEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contact); if (!isPhone && !isEmail) return fail_('답변받으실 연락처(010-1234-5678) 또는 이메일을 정확히 입력해 주세요.'); if (body.length < 5) return fail_('문의 내용을 5자 이상 입력해 주세요.'); const lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (e) { return fail_('접속자가 많아 처리하지 못했습니다. 잠시 후 다시 시도해 주세요.'); } try { appendRow_(INQUIRY_SHEET_NAME, INQUIRY_HEADERS, [name, contact, body, '미처리']); } catch (err) { console.error(err); return fail_('저장 중 오류가 발생했습니다. 잠시 후 다시 시도해 주세요.'); } finally { lock.releaseLock(); } notify_('새 문의가 접수되었습니다.'); return { ok: true }; } /* ───────────── 관리자 조회 ───────────── */ function adminList_(key) { const stored = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY'); if (!stored || stored.length < 16) { return { ok: false, code: 'not_configured', message: '관리자 키가 설정되지 않았습니다. 스크립트 속성에 ADMIN_KEY(16자 이상)를 등록해 주세요.', }; } if (!safeEqual_(String(key == null ? '' : key), stored)) { return { ok: false, code: 'unauthorized', message: '관리자 키가 올바르지 않습니다.' }; } return { ok: true, applications: readSheet_(SHEET_NAME, HEADERS), inquiries: readSheet_(INQUIRY_SHEET_NAME, INQUIRY_HEADERS), }; } function readSheet_(sheetName, headers) { const sheet = getSS_().getSheetByName(sheetName); if (!sheet || sheet.getLastRow() < 2) return { headers: headers, rows: [] }; const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getDisplayValues(); return { headers: headers, rows: values.reverse() }; // 최신순 } /* ───────────── 공통 ───────────── */ function getSS_() { return SPREADSHEET_ID ? SpreadsheetApp.openById(SPREADSHEET_ID) : SpreadsheetApp.getActiveSpreadsheet(); } // 접수일시를 맨 앞에 붙여 한 행을 추가한다. 텍스트 서식이라 앞자리 0이 보존되고 수식이 실행되지 않는다. function appendRow_(sheetName, headers, values) { const sheet = getSheet_(sheetName, headers); const row = sheet.getLastRow() + 1; const stamp = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss'); const range = sheet.getRange(row, 1, 1, headers.length); range.setNumberFormat('@'); range.setValues([[stamp].concat(values)]); SpreadsheetApp.flush(); } function getSheet_(sheetName, headers) { const ss = getSS_(); let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); sheet.setFrozenRows(1); } return sheet; } function notify_(message) { if (!NOTIFY_EMAIL) return; try { MailApp.sendEmail(NOTIFY_EMAIL, '[노즈가드 임상연구] ' + message, message + ' 구글 시트에서 확인해 주세요.'); } catch (err) { console.error(err); // 알림 실패가 접수 자체를 막지 않도록 한다. } } function clean_(v, max) { return String(v == null ? '' : v).replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, max); } function fail_(message) { return { ok: false, message: message }; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } // 길이·내용과 무관하게 끝까지 비교(응답 시간 차이로 키를 추측하는 것을 어렵게 함) function safeEqual_(a, b) { let diff = a.length ^ b.length; const n = Math.max(a.length, b.length); for (let i = 0; i < n; i++) { diff |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0); } return diff === 0; }
